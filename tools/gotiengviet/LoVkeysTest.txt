    // Helper method to load the local file before each test
    private async Task NavigateToAppAsync()
    {
        // Use AppContext.BaseDirectory to reliably point to the bin/Debug/net10.0 folder
        string filePath = Path.Combine(AppContext.BaseDirectory, "web", "inline", "loc-vn-keys.html");

        // Use standard Uri class to automatically format it into a proper file:// URL
        string fileUrl = new Uri(filePath).AbsoluteUri;

        await Page.GotoAsync(fileUrl);

        // Ensure textarea is ready and focused
        await Page.Locator("#vnArea").ClickAsync();
    }

    [Theory]
    // Single characters and circumflex/breve
    [InlineData("aa", "â")]
    [InlineData("aw", "ă")]
    [InlineData("ee", "ê")]
    [InlineData("oo", "ô")]
    [InlineData("ow", "ơ")]
    [InlineData("uw", "ư")]
    [InlineData("dd", "đ")]

    [InlineData("AW", "Ă")]
    [InlineData("DD", "Đ")]

    {
        await NavigateToAppAsync();
        var textArea = Page.Locator("#vnArea");

        // Use PressSequentiallyAsync to trigger `beforeinput` events
        await textArea.PressSequentiallyAsync(input);

        // Verify the value in the text area
        await Expect(textArea).ToHaveValueAsync(expectedOutput);

                switch (currentMode) {

                        break;

                    case 'VNI':

            function onBeforeInput(event) {
                if (
                    currentMode === 'OFF' ||
                    event.inputType !== 'insertText' ||

                    !event.cancelable ||
